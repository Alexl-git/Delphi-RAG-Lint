  Note: `NodeStr` is a local helper replicated in each checker (project convention).

  ```pascal
  class function TAstChecker.CheckGlobalFormVars(const AFile: string): TArray<TLintFinding>;
  var
    Src           : TBytes;
    Parser        : TTSParser;
    Tree          : TTSTree;
    Findings      : TList<TLintFinding>;
    FormClassNames: TDictionary<string, Boolean>;

    function NodeStr(const ANode: TTSNode): string;
    var B: TBytes;
    begin
      if ANode.IsNull or (ANode.StartByte >= ANode.EndByte) then Exit('');
      SetLength(B, Integer(ANode.EndByte) - Integer(ANode.StartByte));
      Move(Src[ANode.StartByte], B[0], Length(B));
      Result:= TEncoding.UTF8.GetString(B);
    end;

    { Pass 1: collect unit-level class type names.
      Skips into defProc/defFunc subtrees only to find nested types (rare) -- but
      we skip the entire proc body to avoid picking up local class-type helpers. }
    procedure CollectClassNames(const N: TTSNode; const AInProc: Boolean);
    var
      I       : Integer;
      Ch      : TTSNode;
      NmN     : TTSNode;
      HasClass: Boolean;
      ClsName : string;
    begin
      if N.IsNull then Exit;
      if (N.NodeType = 'defProc') or (N.NodeType = 'defFunc') then Exit;
      if N.NodeType = 'declType' then
      begin
        NmN:= Default(TTSNode);
        for I:= 0 to N.NamedChildCount - 1 do
          if N.NamedChild(I).NodeType = 'identifier' then
          begin
            NmN:= N.NamedChild(I);
            Break;
          end;
        HasClass:= False;
        for I:= 0 to N.NamedChildCount - 1 do
        begin
          Ch:= N.NamedChild(I);
          if Ch.NodeType = 'declClass' then begin HasClass:= True; Break; end;
        end;
        if HasClass and (not NmN.IsNull) then
        begin
          ClsName:= LowerCase(NodeStr(NmN));
          if ClsName <> '' then FormClassNames.AddOrSetValue(ClsName, True);
        end;
        Exit;
      end;
      for I:= 0 to N.NamedChildCount - 1 do
        CollectClassNames(N.NamedChild(I), False);
    end;

    { Pass 2: find global declVars entries whose declared type is a form class. }
    procedure CheckGlobalVarDecls(const N: TTSNode);
    var
      I, J, K           : Integer;
      DV, DVType, NameId: TTSNode;
      VarTypeName, VarName: string;
      TypeStart         : Integer;
      P                 : TTSPoint;
      F                 : TLintFinding;
    begin
      if N.IsNull then Exit;
      { skip all procedure/function bodies -- their vars are local, not global }
      if (N.NodeType = 'defProc') or (N.NodeType = 'defFunc') then Exit;
      if N.NodeType = 'declVars' then
      begin
        for J:= 0 to N.NamedChildCount - 1 do
        begin
          DV:= N.NamedChild(J);
          if DV.NodeType <> 'declVar' then Continue;
          DVType:= DV.ChildByField('type');
          if DVType.IsNull then Continue;
          VarTypeName:= LowerCase(NodeStr(DVType));
          if not FormClassNames.ContainsKey(VarTypeName) then Continue;
          TypeStart:= Integer(DVType.StartByte);
          for K:= 0 to DV.NamedChildCount - 1 do
          begin
            NameId:= DV.NamedChild(K);
            if NameId.NodeType <> 'identifier' then Continue;
            if Integer(NameId.StartByte) >= TypeStart then Continue;
            VarName:= NodeStr(NameId);
            if VarName = '' then Continue;
            P:= NameId.StartPoint;
            F:= Default(TLintFinding);
            F.RuleId  := 'global-form-variable';
            F.Severity:= 'warning';
            F.Message := Format(
              'Global form variable ''%s: %s'' may leak if the form is created more than ' +
              'once. Consider removing the global and creating/freeing the form locally.',
              [VarName, NodeStr(DVType)]);
            F.FilePath := AFile;
            F.StartLine:= Integer(P.Row   ) + 1;
            F.StartCol := Integer(P.Column) + 1;
            F.EndLine  := F.StartLine;
            F.EndCol   := F.StartCol + Length(VarName);
            Findings.Add(F);
          end;
        end;
        Exit;
      end;
      for I:= 0 to N.NamedChildCount - 1 do
        CheckGlobalVarDecls(N.NamedChild(I));
    end;

  begin
    Result:= nil;
    { Only analyse form units -- a sibling .dfm is the authoritative signal. }
    if not TFile.Exists(ChangeFileExt(AFile, '.dfm')) then Exit;
    Src:= TFile.ReadAllBytes(AFile);
    Findings:= TList<TLintFinding>.Create;
    FormClassNames:= TDictionary<string, Boolean>.Create;
    Parser:= TTSParser.Create;
    try
      Parser.Language:= tree_sitter_delphi13;
      Tree:= Parser.ParseBytes(Src,
        function (AByteIndex: UInt32; APosition: TTSPoint; var ABytesRead: UInt32): TBytes
        var Remaining: Integer;
        begin
          Remaining:= Length(Src) - Integer(AByteIndex);
          if Remaining <= 0 then begin ABytesRead:= 0; SetLength(Result, 0); Exit; end;
          SetLength(Result, Remaining);
          Move(Src[AByteIndex], Result[0], Remaining);
          ABytesRead:= Remaining;
        end, TTSInputEncoding.TSInputEncodingUTF8);
      try
        if Tree <> nil then
        begin
          CollectClassNames(Tree.RootNode, False);
          if FormClassNames.Count > 0 then
            CheckGlobalVarDecls(Tree.RootNode);
        end;
        Result:= Findings.ToArray;
      finally
        Tree.Free;
      end;
    finally
      FormClassNames.Free;
      Findings.Free;
      Parser.Free;
    end;
  end;
  ```

- [ ] **Step 2: Add implementation body to ProjectChecks.pas**

  ```pascal
  class function TProjectChecks.CheckUnitMembership(const AStore: ISymbolStore;
    const ALibDbPath: string; const AProjectDprojPath: string): TArray<TLintFinding>;
  var
    AllFileIds: TArray<Int64>;
    UsesArr   : TArray<TUnitUse>;
    Seen      : TDictionary<string, string>;  { norm -> original UnitName }
    Findings  : TList<TLintFinding>;
    FileId    : Int64;
    U         : TUnitUse;
    UnitNorm  : string;
    UnitOrig  : string;
    DprojDir  : string;
    DprPath   : string;
    DCCRefs   : TArray<string>;
    DprUses   : TArray<string>;
    DummyLine : Integer;
    DCCSet    : TDictionary<string, Boolean>;
    DprSet    : TDictionary<string, Boolean>;
    Ref, RefNorm: string;
    OnDisk, InDpr, InDproj: Boolean;
    F         : TLintFinding;
    LibConn   : TFDConnection;
    LibQ      : TFDQuery;

    function NormUnit(const AName: string): string;
    var DotPos: Integer;
    begin
      Result:= LowerCase(AName);
      DotPos:= LastDelimiter('.', Result);
      if DotPos > 0 then Result:= Copy(Result, DotPos + 1, MaxInt);
    end;

    function IsInLibDb(const ANorm: string): Boolean;
    begin
      Result:= False;
      if not Assigned(LibQ) then Exit;
      LibQ.Close;
      LibQ.Params[0].AsString:= ANorm;
      LibQ.Open;
      Result:= not LibQ.IsEmpty;
    end;

  begin
    Result:= nil;
    Findings:= TList<TLintFinding>.Create;
    Seen    := TDictionary<string, string>.Create;
    DCCSet  := TDictionary<string, Boolean>.Create;
    DprSet  := TDictionary<string, Boolean>.Create;
    LibConn := nil;
    LibQ    := nil;
    try
      { Open library DB if provided }
      if (ALibDbPath <> '') and TFile.Exists(ALibDbPath) then
      begin
        LibConn:= TFDConnection.Create(nil);
        LibConn.DriverName:= 'SQLite';
        LibConn.Params.Values['Database']:= ALibDbPath;
        LibConn.Params.Values['OpenMode']:= 'ReadOnly';
        LibConn.Connected:= True;
        LibQ:= TFDQuery.Create(nil);
        LibQ.Connection:= LibConn;
        LibQ.SQL.Text:= 'SELECT 1 FROM symbols WHERE unit_name_norm=:N LIMIT 1';
        LibQ.Prepare;
      end;

      { Collect all unique used unit names across all project files }
      AllFileIds:= AStore.GetAllFileIds;
      for FileId in AllFileIds do
      begin
        UsesArr:= AStore.GetUnitUsesForFile(FileId);
        for U in UsesArr do
        begin
          UnitNorm:= NormUnit(U.UnitName);
          if (UnitNorm <> '') and not Seen.ContainsKey(UnitNorm) then
            Seen.Add(UnitNorm, U.UnitName);
        end;
      end;

      { Build project membership sets from .dproj + .dpr }
      DprojDir:= '';
      DprPath := '';
      DCCRefs := nil;
      DprUses := nil;
      if (AProjectDprojPath <> '') and TFile.Exists(AProjectDprojPath) then
      begin
        DprojDir:= ExtractFilePath(AProjectDprojPath);
        var DprFiles:= TDirectory.GetFiles(DprojDir, '*.dpr');
        if Length(DprFiles) > 0 then DprPath:= DprFiles[0];
        DCCRefs:= ReadDCCReferences(AProjectDprojPath);
        if DprPath <> '' then
          DprUses:= ExtractUsesNames(DprPath, DummyLine);
        for Ref in DCCRefs do
          DCCSet.AddOrSetValue(NormUnit(ChangeFileExt(ExtractFileName(Ref), '')), True);
        for Ref in DprUses do
          DprSet.AddOrSetValue(NormUnit(Ref), True);
      end;

      { Check each used unit }
      for UnitNorm in Seen.Keys do
      begin
        UnitOrig:= Seen[UnitNorm];
        { Skip built-ins that are never in any index }
        if SameText(UnitNorm, 'system') or SameText(UnitNorm, 'sysinit') then Continue;
        { 1. Library DB check }
        if IsInLibDb(UnitNorm) then Continue;
        { 2. Project membership check }
        if DprojDir <> '' then
        begin
          OnDisk:= TFile.Exists(TPath.Combine(DprojDir, UnitOrig + '.pas'));
          InDpr  := DprSet.ContainsKey(UnitNorm);
          InDproj:= DCCSet.ContainsKey(UnitNorm);
          if OnDisk and InDpr and InDproj then Continue;
        end;
        F:= Default(TLintFinding);
        F.RuleId  := 'unit-not-in-project';
        F.Severity:= 'warning';
        F.Message := Format(
          'Unit ''%s'' is used but not in the platform library and not fully ' +
          'registered in the project (.dpr + .dproj).', [UnitOrig]);
        F.FilePath := AProjectDprojPath;
        F.StartLine:= 0;
        F.StartCol := 0;
        F.EndLine  := 0;
        F.EndCol   := 0;
        Findings.Add(F);
      end;
      Result:= Findings.ToArray;
    finally
      LibQ.Free;
      LibConn.Free;
      DprSet.Free;
      DCCSet.Free;
      Seen.Free;
      Findings.Free;
    end;
  end;
  ```

  ```pascal
  // v0.61: drag-lint lint-all --project <.dproj> [--out <file>] [--json]
  //   [--rule <id>] [--disable <id,...>] [--db <path>]
  // Runs all lint rules (per-file + project-wide) over all .dproj members.
  // Writes a report to --out (default: <proj-dir>\lint-report-YYYYMMDD.txt).
  // Exit 1 if any error-severity finding, 2 on usage error, 0 otherwise.
  function DoLintAll(const AArgs: TArgs): Integer;
  var
    Members    : TArray<string>;
    AllFindings: TArray<TLintFinding>;
    F          : TLintFinding;
    Store      : ISymbolStore;
    ProjDir    : string;
    OutPath    : string;
    Sb         : TStringBuilder;
    JArr       : TJSONArray;
    JObj       : TJSONObject;
    ErrCount   : Integer;
    WarnCount  : Integer;
    InfoCount  : Integer;
    MemberPath : string;
    Dedup      : TDictionary<string, Boolean>;
    DedupKey   : string;
    FileArgs   : TArgs;
    LibDbPath  : string;

    { Sort by severity (error first), then file, then line. }
    procedure SortFindings;
    var
      I, J: Integer;
      Tmp : TLintFinding;

      function SevOrder(const S: string): Integer;
      begin
        if S = 'error'   then Result:= 0
        else if S = 'warning' then Result:= 1
        else if S = 'hint'    then Result:= 2
        else Result:= 3;
      end;

    begin
      for I:= 0 to High(AllFindings) - 1 do
        for J:= I + 1 to High(AllFindings) do
        begin
          var Cmp: Integer:= SevOrder(AllFindings[I].Severity) - SevOrder(AllFindings[J].Severity);
          if Cmp = 0 then Cmp:= CompareStr(AllFindings[I].FilePath, AllFindings[J].FilePath);
          if Cmp = 0 then Cmp:= AllFindings[I].StartLine - AllFindings[J].StartLine;
          if Cmp > 0 then begin Tmp:= AllFindings[I]; AllFindings[I]:= AllFindings[J]; AllFindings[J]:= Tmp; end;
        end;
    end;

  begin
    if AArgs.ProjectPath = '' then
    begin
      Writeln('Usage: drag-lint lint-all --project <.dproj> [--out <file>] [--json] [--rule <id>] [--db <path>]');
      Exit(2);
    end;
    if not TFile.Exists(AArgs.ProjectPath) then
    begin
      Writeln(Format('Project file not found: %s', [AArgs.ProjectPath]));
      Exit(2);
    end;
    ProjDir:= ExtractFilePath(AArgs.ProjectPath);

    { Enumerate .pas members from .dproj }
    var RawRefs:= DRagLint.Lint.ProjectChecks.ReadDCCReferences(AArgs.ProjectPath);
    Members:= nil;
    for var Ref in RawRefs do
    begin
      var AbsPath:= Ref;
      if not TPath.IsPathRooted(AbsPath) then
        AbsPath:= TPath.GetFullPath(TPath.Combine(ProjDir, AbsPath));
      if TFile.Exists(AbsPath) then Members:= Members + [AbsPath];
    end;
    if Length(Members) = 0 then
    begin
      Writeln(Format('No .pas members found in %s', [AArgs.ProjectPath]));
      Exit(2);
    end;

    { Resolve DBs (Two-DB model -- fills AArgs.DbPath + AArgs.DbPaths) }
    var ResolvedArgs:= AArgs;
    ResolveConsumerDbs(ResolvedArgs);
    if not TFile.Exists(ResolvedArgs.DbPath) then
    begin
      Writeln(Format('Project DB not found: %s (pass --db or run from a directory covered by the manifest)', [ResolvedArgs.DbPath]));
      Exit(2);
    end;
    Store:= TSQLiteSymbolStore.Create(ResolvedArgs.DbPath);
    Store.Migrate;
    LibDbPath:= '';
    if Length(ResolvedArgs.DbPaths) > 1 then LibDbPath:= ResolvedArgs.DbPaths[1];

    AllFindings:= nil;

    { Per-file lint: run all TAstChecker rules for each member }
    for MemberPath in Members do
    begin
      FileArgs:= ResolvedArgs;
      FileArgs.Path:= MemberPath;
      { mirror the DoLint checker invocations }
      AllFindings:= AllFindings + DRagLint.Diagnostics.AstChecks.TAstChecker.CheckSyntaxErrors(MemberPath);
      AllFindings:= AllFindings + DRagLint.Diagnostics.AstChecks.TAstChecker.CheckUnbalancedBeginEnd(MemberPath);
      AllFindings:= AllFindings + DRagLint.Diagnostics.AstChecks.TAstChecker.CheckRaiseInFinally(MemberPath);
      AllFindings:= AllFindings + DRagLint.Diagnostics.AstChecks.TAstChecker.CheckCodeAfterExit(MemberPath);
      AllFindings:= AllFindings + DRagLint.Diagnostics.AstChecks.TAstChecker.CheckControlFlowInFinally(MemberPath);
      AllFindings:= AllFindings + DRagLint.Diagnostics.AstChecks.TAstChecker.CheckMissingInherited(MemberPath);
      AllFindings:= AllFindings + DRagLint.Diagnostics.AstChecks.TAstChecker.CheckRoutineMetrics(MemberPath, 7, 25, 120, 5);
      AllFindings:= AllFindings + DRagLint.Diagnostics.AstChecks.TAstChecker.CheckTypeAware(MemberPath);
      AllFindings:= AllFindings + DRagLint.Diagnostics.AstChecks.TAstChecker.CheckFireDacSqlMismatch(MemberPath);
      AllFindings:= AllFindings + DRagLint.Diagnostics.AstChecks.TAstChecker.CheckUnprotectedFree(MemberPath);
      AllFindings:= AllFindings + DRagLint.Diagnostics.AstChecks.TAstChecker.CheckUseAfterFree(MemberPath);
      AllFindings:= AllFindings + DRagLint.Diagnostics.AstChecks.TAstChecker.CheckUiThread(MemberPath);
      AllFindings:= AllFindings + DRagLint.Diagnostics.AstChecks.TAstChecker.CheckGlobalFormVars(MemberPath);
      { DB-backed per-file checker }
      AllFindings:= AllFindings + DRagLint.Diagnostics.AstChecks.TAstChecker.Check(Store, MemberPath);
      AllFindings:= AllFindings + DRagLint.Diagnostics.AstChecks.TAstChecker.CheckUnusedLocals(MemberPath);
    end;

    { Project-wide lint }
    AllFindings:= AllFindings + DRagLint.Lint.ProjectRules.TProjectLintRules.Run(Store, '');
    { unit-not-in-project }
    AllFindings:= AllFindings + DRagLint.Lint.ProjectChecks.TProjectChecks.CheckUnitMembership(
      Store, LibDbPath, ResolvedArgs.ProjectPath);

    { Apply line suppressions and --disable list }
    AllFindings:= ApplyLineSuppressions(AllFindings);
    if ResolvedArgs.Disable <> '' then
    begin
      var DisabledIds: TArray<string>:= ResolvedArgs.Disable.Split([',', ' ', ';']);
      var KeptF: TArray<TLintFinding>:= nil;
      var Drop: Boolean;
      var DId: string;
      for F in AllFindings do
      begin
        Drop:= False;
        for DId in DisabledIds do
          if SameText(Trim(DId), F.RuleId) then begin Drop:= True; Break; end;
        if not Drop then KeptF:= KeptF + [F];
      end;
      AllFindings:= KeptF;
    end;

    { Deduplicate by (file, line, rule, message) }
    Dedup:= TDictionary<string, Boolean>.Create;
    try
      var Unique: TArray<TLintFinding>:= nil;
      for F in AllFindings do
      begin
        DedupKey:= Format('%s|%d|%s|%s', [F.FilePath, F.StartLine, F.RuleId, F.Message]);
        if not Dedup.ContainsKey(DedupKey) then
        begin
          Dedup.Add(DedupKey, True);
          Unique:= Unique + [F];
        end;
      end;
      AllFindings:= Unique;
    finally
      Dedup.Free;
    end;

    SortFindings;

    { Filter by --rule if specified }
    if ResolvedArgs.Rule <> '' then
    begin
      var Filtered: TArray<TLintFinding>:= nil;
      for F in AllFindings do
        if SameText(F.RuleId, ResolvedArgs.Rule) then Filtered:= Filtered + [F];
      AllFindings:= Filtered;
    end;

    { Count by severity }
    ErrCount := 0;
    WarnCount := 0;
    InfoCount := 0;
    for F in AllFindings do
      if F.Severity = 'error'   then Inc(ErrCount)
      else if F.Severity = 'warning' then Inc(WarnCount)
      else Inc(InfoCount);

    { Determine output path }
    OutPath:= ResolvedArgs.Output;
    if OutPath = '' then
      OutPath:= TPath.Combine(ProjDir, Format('lint-report-%s.txt', [FormatDateTime('YYYYMMDD', Now)]));

    { Build and write output }
    if ResolvedArgs.AsJson then
    begin
      JArr:= TJSONArray.Create;
      try
        for F in AllFindings do
        begin
          JObj:= TJSONObject.Create;
          JObj.AddPair('rule'      , F.RuleId  );
          JObj.AddPair('severity'  , F.Severity);
          JObj.AddPair('file_path' , F.FilePath);
          JObj.AddPair('start_line', TJSONNumber.Create(F.StartLine));
          JObj.AddPair('start_col' , TJSONNumber.Create(F.StartCol ));
          JObj.AddPair('end_line'  , TJSONNumber.Create(F.EndLine  ));
          JObj.AddPair('end_col'   , TJSONNumber.Create(F.EndCol   ));
          JObj.AddPair('message'   , F.Message );
          JArr.AddElement(JObj);
        end;
        var JsonRoot:= TJSONObject.Create;
        try
          JsonRoot.AddPair('project'  , AArgs.ProjectPath);
          JsonRoot.AddPair('generated', FormatDateTime('yyyy-mm-dd"T"hh:nn:ss', Now));
          JsonRoot.AddPair('findings' , JArr);
          TFile.WriteAllText(OutPath, JsonRoot.Format(2), TEncoding.UTF8);
          Writeln(JsonRoot.Format(2));
        finally
          JsonRoot.Free;
        end;
      finally
        { JArr now owned by JsonRoot -- already freed above }
      end;
    end
    else
    begin
      Sb:= TStringBuilder.Create;
      try
        for F in AllFindings do
          Sb.AppendLine(Format('%s(%d,%d): [%s] %s', [F.FilePath, F.StartLine, F.StartCol, F.RuleId, F.Message]));
        Sb.AppendLine(Format('%d finding(s) -- %d error(s), %d warning(s), %d info',
          [Length(AllFindings), ErrCount, WarnCount, InfoCount]));
        TFile.WriteAllText(OutPath, Sb.ToString, TEncoding.UTF8);
        for F in AllFindings do
          Writeln(Format('%s(%d,%d): [%s] %s', [F.FilePath, F.StartLine, F.StartCol, F.RuleId, F.Message]));
      finally
        Sb.Free;
      end;
    end;

    Writeln(Format('Report written to: %s', [OutPath]));
    Writeln(Format('%d finding(s) -- %d error(s), %d warning(s), %d info',
      [Length(AllFindings), ErrCount, WarnCount, InfoCount]));

    if ErrCount > 0 then Result:= 1 else Result:= 0;
  end;
  ```

  **Note:** The `JArr` double-free issue above: when `JArr` is added via `JsonRoot.AddPair('findings', JArr)`, the root TJSONObject takes ownership. Remove the inner `finally JArr.Free` — instead, just let `JsonRoot.Free` handle it. The code above already does this correctly (the outer `try..finally` uses `JsonRoot.Free` which frees `JArr` via ownership).
